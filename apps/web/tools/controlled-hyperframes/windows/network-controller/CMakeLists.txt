cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(CourseforgeNetworkPolicy LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Requires an explicitly provisioned Windows MSVC toolchain and Windows SDK")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only the explicitly pinned x64 host is supported")
endif()
add_library(courseforge_network_policy STATIC WfpNetworkPolicy.cpp)
target_compile_features(courseforge_network_policy PUBLIC cxx_std_17)
target_compile_definitions(courseforge_network_policy PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(courseforge_network_policy PRIVATE /W4 /WX /permissive-)
set_property(TARGET courseforge_network_policy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(courseforge_network_policy PRIVATE /Brepro)
target_link_libraries(courseforge_network_policy PUBLIC Fwpuclnt Advapi32 Rpcrt4 Uuid)
add_library(courseforge_network_bridge SHARED NetworkPolicyExports.cpp)
target_compile_features(courseforge_network_bridge PRIVATE cxx_std_17)
target_compile_options(courseforge_network_bridge PRIVATE /W4 /WX /permissive-)
set_property(TARGET courseforge_network_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(courseforge_network_bridge PRIVATE /Brepro)
target_link_options(courseforge_network_bridge PRIVATE /Brepro /INCREMENTAL:NO)
target_link_libraries(courseforge_network_bridge PRIVATE courseforge_network_policy)
# Libraries only: no executable, service, installation, activation or filter invocation.
